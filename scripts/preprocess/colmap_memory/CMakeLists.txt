cmake_minimum_required(VERSION 3.18)
project(pgsr_colmap_memory LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(colmap REQUIRED)
if(NOT COLMAP_VERSION STREQUAL "3.9.1")
  message(FATAL_ERROR "This memory patch requires COLMAP 3.9.1")
endif()
add_executable(mapper-memory mapper.cc database_cache.cc correspondence_graph.cc)
target_link_libraries(mapper-memory PRIVATE /usr/local/lib/libcolmap_exe.a colmap::colmap)
add_executable(probe-stock probe.cc)
target_link_libraries(probe-stock PRIVATE colmap::colmap)
add_executable(probe-memory probe.cc database_cache.cc correspondence_graph.cc)
target_link_libraries(probe-memory PRIVATE colmap::colmap)
